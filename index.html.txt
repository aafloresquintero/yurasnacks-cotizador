<script>
    let mensajeWhatsApp = "";

    function calcularTotal() {
        const selectProducto = document.getElementById('producto');
        const precioUnitario = parseFloat(selectProducto.value);
        const nombreProducto = selectProducto.options[selectProducto.selectedIndex].getAttribute('data-nombre');
        
        let inputCantidad = document.getElementById('cantidad');
        let cantidad = parseInt(inputCantidad.value);
        
        // Blindaje anti-errores: Si borran el campo o ponen un número menor a 1, lo forzamos a 1
        if (isNaN(cantidad) || cantidad < 1) {
            cantidad = 1;
            inputCantidad.value = 1;
        }
        
        const radioEntrega = document.querySelector('input[name="entrega"]:checked');
        const costoEntrega = radioEntrega ? parseFloat(radioEntrega.value) : 0;
        const nombreEntrega = costoEntrega > 0 ? "Delivery en Lima" : "Recojo en punto de venta";

        const subtotal = precioUnitario * cantidad;
        const total = subtotal + costoEntrega;

        document.getElementById('textoTotal').innerText = `Total: S/ ${total.toFixed(2)}`;
        
        // Formato exacto solicitado
        mensajeWhatsApp = `Pedido: ${cantidad} x ${nombreProducto}, canal: ${nombreEntrega}, total: S/ ${total.toFixed(2)}`;
        
        document.getElementById('textoResumen').innerText = `Detalle: ${cantidad}x ${nombreProducto}\nCanal: ${nombreEntrega}`;

        document.getElementById('seccionResultado').style.display = 'block';
        document.getElementById('btnCopiar').style.display = 'block';
        document.getElementById('btnNuevo').style.display = 'block';
    }

    function copiarResumen() {
        if (!mensajeWhatsApp) return;
        
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(mensajeWhatsApp).then(() => {
                feedbackCopiado();
            }).catch(() => {
                fallbackCopiarTexto(mensajeWhatsApp);
            });
        } else {
            fallbackCopiarTexto(mensajeWhatsApp);
        }
    }

    function feedbackCopiado() {
        const btn = document.getElementById('btnCopiar');
        const textoOriginal = btn.innerText;
        btn.innerText = "¡Copiado al portapapeles! ✓";
        btn.style.backgroundColor = "#a3b19b";
        
        setTimeout(() => {
            btn.innerText = textoOriginal;
            btn.style.backgroundColor = "#d8c3a5";
        }, 2000);
    }

    function fallbackCopiarTexto(texto) {
        let textarea = document.createElement("textarea");
        textarea.value = texto;
        document.body.appendChild(textarea);
        textarea.select();
        try {
            document.execCommand('copy');
            feedbackCopiado();
        } catch (err) {
            alert("No se pudo copiar automáticamente. Selecciónalo y cópialo manualmente: \n\n" + texto);
        }
        document.body.removeChild(textarea);
    }

    function reiniciarCotizador() {
        document.getElementById('cantidad').value = "1";
        document.getElementById('producto').selectedIndex = 0;
        document.querySelector('input[name="entrega"][value="0"]').checked = true;
        document.getElementById('seccionResultado').style.display = 'none';
        document.getElementById('btnCopiar').style.display = 'none';
        document.getElementById('btnNuevo').style.display = 'none';
        mensajeWhatsApp = "";
    }
</script>
